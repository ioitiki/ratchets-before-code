```rust
//! Wire types for the ledger: everything that crosses a process boundary.
//!
//! This crate holds data, not decisions. The only behavior here is what a
//! type must say about itself, such as which effect semantics a job kind
//! declares (ADR 0003).

use serde::{Deserialize, Serialize};

/// A point in time, in milliseconds. The core never reads a clock: whoever
/// calls it says what time it is (ADR 0002).
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct Millis(pub u64);

impl Millis {
    /// `self` plus `ms`, saturating instead of overflowing.
    pub fn plus(self, ms: u64) -> Millis {
        Millis(self.0.saturating_add(ms))
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct JobId(pub u64);

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
pub struct LeaseId(pub u64);

/// What a job does. The demo has one kind.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum JobKind {
    /// Sleep for `ms`, then record `token` as the job's result.
    SleepThenRecord { ms: u64, token: String },
}

/// What recovery may assume about re-running an effect whose outcome was
/// never recorded (ADR 0003).
#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum EffectSemantics {
    /// Running it twice has the same result as running it once: re-drive it.
    Idempotent,
    /// Running it twice may do harm: never re-run it silently.
    NotIdempotent,
}

impl JobKind {
    /// Every kind declares its semantics here, where it is defined, because
    /// they cannot be worked out after a crash.
    pub fn effect_semantics(&self) -> EffectSemantics {
        match self {
            JobKind::SleepThenRecord { .. } => EffectSemantics::Idempotent,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum JobState {
    /// Waiting to be leased, no earlier than `not_before`.
    Pending { not_before: Millis },
    /// Held by a worker until `expires`.
    Leased { lease: LeaseId, expires: Millis },
    Succeeded { result: String },
    Dead { reason: String },
}

#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct Job {
    pub id: JobId,
    pub idempotency_key: String,
    pub kind: JobKind,
    pub max_attempts: u32,
    /// Attempts started so far. A lease starts an attempt.
    pub attempts: u32,
    pub state: JobState,
}

/// A request to change the ledger. The core turns each one into events or a
/// refusal.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum Command {
    Submit { idempotency_key: String, kind: JobKind, max_attempts: u32 },
    Lease { now: Millis, lease_for_ms: u64 },
    Complete { job: JobId, lease: LeaseId, result: String },
    Fail { job: JobId, lease: LeaseId, reason: String, now: Millis },
    ExpireLeases { now: Millis },
}

/// A fact the ledger has recorded. State only ever changes by applying events.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum Event {
    Submitted { job: JobId, idempotency_key: String, kind: JobKind, max_attempts: u32 },
    Leased { job: JobId, lease: LeaseId, attempt: u32, expires: Millis },
    Succeeded { job: JobId, result: String },
    Retrying { job: JobId, not_before: Millis, reason: String },
    Died { job: JobId, reason: String },
}

/// Why the core declined a command. A refusal records nothing.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub enum Refusal {
    MaxAttemptsMustBePositive,
    UnknownJob { job: JobId },
    /// The lease is not the job's current lease, or the job is not leased.
    StaleLease { job: JobId, lease: LeaseId },
}
```

```rust
//! Worked examples of every rule the core enforces.

use ledger_api::{Command, Event, JobId, JobKind, JobState, LeaseId, Millis, Refusal};
use ledger_core::{State, apply, decide};

fn submit(key: &str, max_attempts: u32) -> Command {
    Command::Submit {
        idempotency_key: key.into(),
        kind: JobKind::SleepThenRecord { ms: 1, token: key.into() },
        max_attempts,
    }
}

/// Decide, then apply what was decided: what the server will do with every command.
fn run(state: &mut State, command: Command) -> Result<Vec<Event>, Refusal> {
    let events = decide(state, command)?;
    for event in &events {
        apply(state, event);
    }
    Ok(events)
}
```

```rust
//! The ledger's state. It is rebuilt only by applying events.

use std::collections::BTreeMap;

use ledger_api::{Event, Job, JobId, JobState, LeaseId, Millis};

#[derive(Debug, Clone, Default, PartialEq, Eq)]
pub struct State {
    jobs: BTreeMap<JobId, Job>,
    by_key: BTreeMap<String, JobId>,
    next_job: u64,
    next_lease: u64,
}

impl State {
    pub fn job(&self, id: JobId) -> Option<&Job> {
        self.jobs.get(&id)
    }

    pub fn job_for_key(&self, key: &str) -> Option<JobId> {
        self.by_key.get(key).copied()
    }

    pub fn jobs(&self) -> impl Iterator<Item = &Job> {
        self.jobs.values()
    }

    pub(crate) fn next_job_id(&self) -> JobId {
        JobId(self.next_job)
    }

    pub(crate) fn next_lease_id(&self) -> LeaseId {
        LeaseId(self.next_lease)
    }

    /// The pending job that has been due longest at `now`. Ties go to the lowest id.
    pub(crate) fn next_due(&self, now: Millis) -> Option<&Job> {
        self.jobs
            .values()
            .filter_map(|job| match job.state {
                JobState::Pending { not_before } if not_before <= now => Some((not_before, job)),
                _ => None,
            })
            .min_by_key(|(not_before, job)| (*not_before, job.id))
            .map(|(_, job)| job)
    }

    pub(crate) fn expired_leases(&self, now: Millis) -> impl Iterator<Item = &Job> {
        self.jobs
            .values()
            .filter(move |job| matches!(job.state, JobState::Leased { expires, .. } if expires <= now))
    }

    pub(crate) fn apply(&mut self, event: &Event) {
        match event {
            Event::Submitted { job, idempotency_key, kind, max_attempts } => {
                self.jobs.insert(
                    *job,
                    Job {
                        id: *job,
                        idempotency_key: idempotency_key.clone(),
                        kind: kind.clone(),
                        max_attempts: *max_attempts,
                        attempts: 0,
                        state: JobState::Pending { not_before: Millis(0) },
                    },
                );
                self.by_key.insert(idempotency_key.clone(), *job);
                self.next_job = self.next_job.max(job.0 + 1);
            }
            Event::Leased { job, lease, attempt, expires } => {
                if let Some(found) = self.jobs.get_mut(job) {
                    found.attempts = *attempt;
                    found.state = JobState::Leased { lease: *lease, expires: *expires };
                }
                self.next_lease = self.next_lease.max(lease.0 + 1);
            }
            Event::Succeeded { job, result } => {
                self.set_state(*job, JobState::Succeeded { result: result.clone() });
            }
            Event::Retrying { job, not_before, .. } => {
                self.set_state(*job, JobState::Pending { not_before: *not_before });
            }
            Event::Died { job, reason } => {
                self.set_state(*job, JobState::Dead { reason: reason.clone() });
            }
        }
    }

    fn set_state(&mut self, job: JobId, state: JobState) {
        if let Some(found) = self.jobs.get_mut(&job) {
            found.state = state;
        }
    }
}
```

```rust
//! The ledger's decisions, as pure functions.
//!
//! `decide` turns a command into events, or a refusal. `apply` folds an event
//! into state. Nothing here performs I/O, reads a clock or draws randomness;
//! `scripts/ratchets/core-purity.sh` fails the build if that changes.

mod backoff;
mod state;

pub use backoff::backoff_ms;
pub use state::State;

use ledger_api::{Command, Event, Job, JobId, JobState, LeaseId, Millis, Refusal};

/// Decide what a command means against the current state.
///
/// Returns the events to record. An empty list means the command was valid
/// but changes nothing: a duplicate submission, or a lease request when no
/// job is due.
pub fn decide(state: &State, command: Command) -> Result<Vec<Event>, Refusal> {
    match command {
        Command::Submit { idempotency_key, kind, max_attempts } => {
            if max_attempts == 0 {
                return Err(Refusal::MaxAttemptsMustBePositive);
            }
            if state.job_for_key(&idempotency_key).is_some() {
                return Ok(vec![]);
            }
            let job = state.next_job_id();
            Ok(vec![Event::Submitted { job, idempotency_key, kind, max_attempts }])
        }
        Command::Lease { now, lease_for_ms } => Ok(state
            .next_due(now)
            .map(|job| Event::Leased {
                job: job.id,
                lease: state.next_lease_id(),
                attempt: job.attempts + 1,
                expires: now.plus(lease_for_ms),
            })
            .into_iter()
            .collect()),
        Command::Complete { job, lease, result } => {
            holding(state, job, lease)?;
            Ok(vec![Event::Succeeded { job, result }])
        }
        Command::Fail { job, lease, reason, now } => {
            let held = holding(state, job, lease)?;
            Ok(vec![after_failed_attempt(held, reason, now)])
        }
        Command::ExpireLeases { now } => Ok(state
            .expired_leases(now)
            .map(|job| after_failed_attempt(job, "lease expired".to_string(), now))
            .collect()),
    }
}

/// Fold one event into state. Events are facts, so applying one never fails.
pub fn apply(state: &mut State, event: &Event) {
    state.apply(event);
}

/// The job, if `lease` is its current lease.
fn holding(state: &State, job: JobId, lease: LeaseId) -> Result<&Job, Refusal> {
    let found = state.job(job).ok_or(Refusal::UnknownJob { job })?;
    match &found.state {
        JobState::Leased { lease: current, .. } if *current == lease => Ok(found),
        _ => Err(Refusal::StaleLease { job, lease }),
    }
}

/// A failed or expired attempt schedules a retry, or kills the job when no
/// attempts are left.
fn after_failed_attempt(job: &Job, reason: String, now: Millis) -> Event {
    if job.attempts >= job.max_attempts {
        Event::Died {
            job: job.id,
            reason: format!("{reason} (after {} of {} attempts)", job.attempts, job.max_attempts),
        }
    } else {
        Event::Retrying { job: job.id, not_before: now.plus(backoff_ms(job.attempts)), reason }
    }
}
```

```rust
//! Properties that must hold for every sequence of commands, not just the
//! examples someone thought of.

use std::collections::BTreeSet;

use ledger_api::{Command, Event, JobId, JobKind, JobState, LeaseId, Millis, Refusal};
use ledger_core::{State, apply, decide};
use proptest::prelude::*;

/// One step of a random scenario. Job and lease references are small numbers,
/// so steps often hit real jobs and, half the time, guess the wrong lease.
#[derive(Debug, Clone)]
enum Step {
    Submit { key: u8, max_attempts: u32 },
    Lease { advance_ms: u64, lease_for_ms: u64 },
    Complete { job: u8, use_current_lease: bool, guess: u64 },
    Fail { job: u8, use_current_lease: bool, guess: u64 },
    Expire { advance_ms: u64 },
}

fn step() -> impl Strategy<Value = Step> {
    prop_oneof![
        (0u8..6, 0u32..4).prop_map(|(key, max_attempts)| Step::Submit { key, max_attempts }),
        (0u64..5_000, 1u64..10_000)
            .prop_map(|(advance_ms, lease_for_ms)| Step::Lease { advance_ms, lease_for_ms }),
        (0u8..8, any::<bool>(), 0u64..8)
            .prop_map(|(job, use_current_lease, guess)| Step::Complete { job, use_current_lease, guess }),
        (0u8..8, any::<bool>(), 0u64..8)
            .prop_map(|(job, use_current_lease, guess)| Step::Fail { job, use_current_lease, guess }),
        (0u64..20_000).prop_map(|advance_ms| Step::Expire { advance_ms }),
    ]
}

fn steps() -> impl Strategy<Value = Vec<Step>> {
    prop::collection::vec(step(), 0..80)
}

/// A scenario in progress: the state, every event recorded, and the clock.
struct Run {
    state: State,
    events: Vec<Event>,
    now: Millis,
}

impl Run {
    fn new() -> Self {
        Run { state: State::default(), events: vec![], now: Millis(0) }
    }

    fn lease_for(&self, job: JobId, use_current: bool, guess: u64) -> LeaseId {
        match self.state.job(job).map(|found| &found.state) {
            Some(JobState::Leased { lease, .. }) if use_current => *lease,
            _ => LeaseId(guess),
        }
    }

    fn command(&mut self, step: &Step) -> Command {
        match *step {
            Step::Submit { key, max_attempts } => Command::Submit {
                idempotency_key: format!("key-{key}"),
                kind: JobKind::SleepThenRecord { ms: 1, token: format!("t-{key}") },
                max_attempts,
            },
            Step::Lease { advance_ms, lease_for_ms } => {
                self.now = self.now.plus(advance_ms);
                Command::Lease { now: self.now, lease_for_ms }
            }
            Step::Complete { job, use_current_lease, guess } => {
                let job = JobId(u64::from(job));
                Command::Complete { job, lease: self.lease_for(job, use_current_lease, guess), result: "done".into() }
            }
            Step::Fail { job, use_current_lease, guess } => {
                let job = JobId(u64::from(job));
                let lease = self.lease_for(job, use_current_lease, guess);
                Command::Fail { job, lease, reason: "boom".into(), now: self.now }
            }
            Step::Expire { advance_ms } => {
                self.now = self.now.plus(advance_ms);
                Command::ExpireLeases { now: self.now }
            }
        }
    }

    /// Decide one step and apply what was decided.
    fn step(&mut self, step: &Step) -> (Command, Result<Vec<Event>, Refusal>) {
        let command = self.command(step);
        let outcome = decide(&self.state, command.clone());
        if let Ok(events) = &outcome {
            for event in events {
                apply(&mut self.state, event);
                self.events.push(event.clone());
            }
        }
        (command, outcome)
    }
}

proptest! {
    #![proptest_config(ProptestConfig::with_cases(512))]

    #[test]
    fn a_job_succeeds_at_most_once(steps in steps()) {
        let mut run = Run::new();
        for step in &steps {
            run.step(step);
        }
        let mut succeeded = BTreeSet::new();
        for event in &run.events {
            if let Event::Succeeded { job, .. } = event {
                prop_assert!(succeeded.insert(*job), "{job:?} succeeded twice");
            }
        }
    }

    #[test]
    fn attempts_never_exceed_max_attempts(steps in steps()) {
        let mut run = Run::new();
        for step in &steps {
            run.step(step);
            for job in run.state.jobs() {
                prop_assert!(job.attempts <= job.max_attempts, "{job:?}");
            }
        }
    }

    #[test]
    fn a_stale_lease_changes_nothing(steps in steps()) {
        let mut run = Run::new();
        for step in &steps {
            let before = run.state.clone();
            let (command, outcome) = run.step(step);
            let held = |job: &JobId, lease: &LeaseId| {
                matches!(before.job(*job).map(|found| &found.state),
                    Some(JobState::Leased { lease: current, .. }) if current == lease)
            };
            let stale = match &command {
                Command::Complete { job, lease, .. } | Command::Fail { job, lease, .. } => !held(job, lease),
                _ => false,
            };
            if stale {
                prop_assert!(outcome.is_err(), "{command:?} was accepted with a stale lease");
                prop_assert_eq!(&run.state, &before);
            }
        }
    }

    #[test]
    fn replaying_the_events_rebuilds_the_state(steps in steps()) {
        let mut run = Run::new();
        for step in &steps {
            run.step(step);
        }
        let mut replayed = State::default();
        for event in &run.events {
            apply(&mut replayed, event);
        }
        prop_assert_eq!(replayed, run.state);
    }

    #[test]
    fn the_same_commands_give_the_same_events(steps in steps()) {
        let (mut first, mut second) = (Run::new(), Run::new());
        for step in &steps {
            first.step(step);
            second.step(step);
        }
        prop_assert_eq!(first.events, second.events);
    }
}
```

```markdown
# ADR 0002: Time is an input to the core

**Status:** accepted
**Date:** 2026-10-14
**Decided by:** Andy

## Context

Retries, backoff and lease expiry are all decisions about time. If
`ledger-core` reads a clock to make them, every test of those decisions has
to control a real clock, either by sleeping or by mocking a global. The
properties that matter most (a job succeeds at most once, a stale lease
changes nothing, replaying events rebuilds the state) would only ever be
tested at whatever moments the test happened to run.

## Decision

The core never reads a clock. Every command that depends on time carries
`now: Millis`. Whoever calls the core, here `ledgerd`, reads the clock once
and passes it in. The same holds for randomness: backoff has no jitter, and if
jitter is ever needed, its entropy arrives as an argument.

`core-purity.sh` enforces this. `SystemTime::now`, `Instant::now` and `rand`
in the core's source fail the build.

## Consequences

- Every time-dependent rule can be tested at any instant, including instants
  no real test run would ever hit. The property tests drive the clock forward
  in random jumps.
- Replaying a journal reproduces the same decisions, because the times are in
  the journal, not in the machine that replays it.
- Clock skew and monotonicity become the caller's problem, in one place,
  instead of the core's problem in every function.
- Commands are slightly larger. That is the whole cost.
```
